#   python3.10 -m venv ~/.neuro-slice-legacy

#   ~/.neuro-slice-legacy/bin/python -m pip install -r legacy_wsl_requirements.txt
#

# - This file is intentionally conservative.
# - The detector runtime is separate from the main Windows project environment.
